~~~tsx
import { useState } from 'react';
import * as R from '@sdr/react';

const locale = 'en';

export function CookieBannerExample() {
  const [choice, setChoice] = useState('');
  return (
    <R.Stack>
      {!choice && (
        <R.CookieBanner
          title="Cookie preferences"
          acceptLabel="Accept analytics"
          rejectLabel="Reject analytics"
          onAccept={() =>
            setChoice('Analytics accepted in this demonstration.')
          }
          onReject={() =>
            setChoice('Analytics rejected in this demonstration.')
          }
        >
          Optional cookies remain off until you choose.
        </R.CookieBanner>
      )}
      <p
        role="status"
        hidden={!choice}
      >
        {choice}
      </p>
    </R.Stack>
  );
}
~~~

## Usage

The host must gate optional scripts and store consent. This component never sets cookies itself.



## API

| Property | Type | Required |
|---|---|---|
| `title` | `string` | Yes |
| `children` | `ReactNode` | Yes |
| `acceptLabel` | `string` | Yes |
| `rejectLabel` | `string` | Yes |
| `onAccept` | `() => void` | Yes |
| `onReject` | `() => void` | Yes |

Native HTML, aria-* and data-* attributes are available where allowed by the component types. Component-specific properties are extracted from TypeScript when generating documentation.
