## Exemplu

Scopul colectării este explicat clar. Întrebările opționale pentru monitorizarea egalității de șanse sunt separate de condițiile de eligibilitate și includ „Prefer să nu răspund”. Categoriile se stabilesc împreună cu comunitățile vizate.

~~~tsx
import { useState } from 'react';
import * as R from '@strictlylucian/sdr-react';

const locale = 'ro';

export function EqualityInformationExample() {
  const [done, setDone] = useState(false);
  return (
    <form
      data-sdr-pattern-demo="equality"
      className="sdr-stack"
      onSubmit={(event) => {
        event.preventDefault();
        const form = event.currentTarget;
        setDone(true);
        requestAnimationFrame(() =>
          form
            .closest('[data-sdr-pattern-demo]')!
            .querySelector<HTMLElement>('[data-sdr-pattern-result]')!
            .focus(),
        );
      }}
    >
      <R.Fieldset
        legend="Grupă de vârstă (opțional)"
        hint="Răspunsul ne ajută să urmărim accesul la servicii și nu afectează eligibilitatea."
      >
        <R.Radio
          name="age-group"
          value="under18"
          label="Sub 18 ani"
        />
        <R.Radio
          name="age-group"
          value="adult"
          label="18–64 de ani"
        />
        <R.Radio
          name="age-group"
          value="older"
          label="65 de ani sau mai mult"
        />
        <R.Radio
          name="age-group"
          value="private"
          label="Prefer să nu răspund"
        />
      </R.Fieldset>
      <div>
        <R.Button type="submit">Continuă</R.Button>
      </div>
      <p
        role="status"
        data-sdr-pattern-result
        tabIndex={-1}
        hidden={!done}
      >
        Răspunsul a fost înregistrat.
      </p>
    </form>
  );
}
~~~
