~~~tsx
import { useState } from 'react';
import * as R from '@sdr/react';

const locale = 'en';

export function PageExample() {
  const [submitted, setSubmitted] = useState(false);
  return (
    <>
      <R.SkipLink
        href="#page-main"
        children="Skip to content"
      />
      <R.SdrProvider
        lang={locale}
        theme="dark"
      >
        <R.Header
          serviceName="Community services"
          href="#page-main"
        >
          <R.LanguageNavigation
            label="Language"
            items={[
              {
                lang: 'ro',
                label: 'Română',
                href: '/examples/ro/page.html',
                current: locale === 'ro',
              },
              {
                lang: 'en',
                label: 'English',
                href: '/examples/en/page.html',
                current: locale === 'en',
              },
            ]}
          />
        </R.Header>
      </R.SdrProvider>
      <R.ServiceNavigation
        label="Main navigation"
        items={[
          { label: 'Request a document', href: '#page-main', current: true },
          { label: 'Required documents', href: '#page-documents' },
          { label: 'Help', href: '#page-help' },
        ]}
        menuLabel="Menu"
      />
      <R.Container>
        <div className="sdr-u-padding-top-4">
          <R.Breadcrumbs
            label="Breadcrumb"
            items={[
              { label: 'Services', href: '#page-main' },
              {
                label: 'Request a document',
                href: '#page-request',
                current: true,
              },
            ]}
          />
        </div>
        <main
          id="page-main"
          tabIndex={-1}
          className="sdr-main-wrapper sdr-stack"
        >
          <R.PageHeader
            className="sdr-u-padding-block-0"
            title="Request a document"
            description="Complete the details below and check your information before submitting."
          />
          <R.Alert title="Demonstration example">
            Use fictional details. This example does not submit or store
            information.
          </R.Alert>
          <div className="sdr-grid-row">
            <section
              className="sdr-grid-column-two-thirds-from-desktop"
              id="page-request"
              aria-labelledby="page-request-title"
            >
              <R.Stack>
                <h2 id="page-request-title">Application details</h2>
                <form
                  className="sdr-stack"
                  data-sdr-page-form
                  onSubmit={(event) => {
                    event.preventDefault();
                    setSubmitted(true);
                    requestAnimationFrame(() =>
                      document.getElementById('page-confirmation')?.focus(),
                    );
                  }}
                >
                  <R.TextInput
                    id="full-name"
                    name="full-name"
                    label="Full name"
                    autoComplete="name"
                    required
                  />
                  <R.TextInput
                    id="email"
                    name="email"
                    label="Email"
                    type="email"
                    autoComplete="email"
                    hint="We will use this address to contact you about the application."
                    required
                  />
                  <R.Select
                    id="document-type"
                    name="document-type"
                    label="Document type"
                    required
                    defaultValue=""
                  >
                    <option value="">Choose a document</option>
                    <option value="certificate">Certificate</option>
                    <option value="copy">Document copy</option>
                  </R.Select>
                  <R.Checkbox
                    id="confirm-details"
                    name="confirm-details"
                    label="I confirm that the details are correct."
                    required
                  />
                  <div>
                    <R.Button type="submit">Submit application</R.Button>
                  </div>
                </form>
                <section
                  id="page-confirmation"
                  data-sdr-page-result
                  hidden={!submitted}
                  tabIndex={-1}
                  aria-label="Demonstration confirmation"
                >
                  <R.Panel title="Example completed">
                    <p>
                      Your details were checked in the browser. No information
                      was submitted or stored.
                    </p>
                  </R.Panel>
                </section>
              </R.Stack>
            </section>
            <aside
              className="sdr-grid-column-one-third-from-desktop sdr-stack"
              aria-label="Service information"
            >
              <R.Card title="Before you start">
                <R.Stack gap={3}>
                  <p>
                    In a real service, this section explains eligibility, cost
                    and the response timeframe.
                  </p>
                  <R.Cluster>
                    <R.Tag>Demonstration</R.Tag>
                  </R.Cluster>
                </R.Stack>
              </R.Card>
              <section
                id="page-documents"
                aria-labelledby="page-documents-title"
              >
                <R.Stack gap={3}>
                  <h2
                    id="page-documents-title"
                    className="sdr-heading-md"
                  >
                    Required documents
                  </h2>
                  <ul>
                    <li>The applicant’s identity document.</li>
                    <li>
                      The original document, if you are requesting a copy.
                    </li>
                    <li>
                      Authorisation, if you are applying on someone else’s
                      behalf.
                    </li>
                  </ul>
                </R.Stack>
              </section>
            </aside>
          </div>
        </main>
      </R.Container>
      <R.Footer className="sdr-u-padding-inline-0">
        <R.Container>
          <R.Stack gap={3}>
            <h2
              id="page-help"
              className="sdr-heading-sm"
            >
              Need help?
            </h2>
            <p>
              A real service provides contact details, opening hours and
              assisted support options here.
            </p>
            <R.Cluster>
              <R.Link href={`/${locale}/accessibility/`}>Accessibility</R.Link>
              <R.Link href={`/${locale}/patterns/cookies/`}>Cookies</R.Link>
            </R.Cluster>
            <p>SDR · Independent proposal · Example with fictional data</p>
          </R.Stack>
        </R.Container>
      </R.Footer>
    </>
  );
}
~~~

## Usage

Use this composition as a starting point. Replace the fictional service content and implement your service validation and submission.



## Composition

Page is a complete example rather than a separate library component. It combines Header, ServiceNavigation, Breadcrumbs, PageHeader, Grid, form fields, Alert and Footer. See those component pages for their APIs.

Native HTML, aria-* and data-* attributes are available where allowed by the component types. Component-specific properties are extracted from TypeScript when generating documentation.



## Accessibility

Composition accessibility depends on labels, element semantics, focus order and application context. Library themes and behaviours are checked against the documented examples; those results do not assess the consuming service. The matrix and limitations are described in [the accessibility guide](/en/accessibility/).
