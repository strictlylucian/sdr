~~~tsx
import { useState } from 'react';
import * as R from '@strictlylucian/sdr-react';

const locale = 'en';

export function PaymentCardExample() {
  const [done, setDone] = useState(false);
  const amount = new Intl.NumberFormat('en-GB', {
    style: 'currency',
    currency: 'RON',
  }).format(25);
  return (
    <form
      data-sdr-pattern-demo="payment"
      className="sdr-stack"
      onSubmit={(event) => {
        event.preventDefault();
        const form = event.currentTarget;
        setDone(true);
        requestAnimationFrame(() =>
          form
            .closest('[data-sdr-pattern-demo]')!
            .querySelector<HTMLElement>('[data-sdr-pattern-result]')!
            .focus(),
        );
      }}
    >
      <h3>Pay the document fee</h3>
      <R.SummaryList items={[{ key: 'Amount due', value: amount }]} />
      <R.TextInput
        label="Card number"
        name="card-number"
        inputMode="numeric"
        autoComplete="cc-number"
        defaultValue="4242 4242 4242 4242"
        pattern="4242(?:\s|\x2D)*4242(?:\s|\x2D)*4242(?:\s|\x2D)*4242"
        title="Use test card 4242 4242 4242 4242."
        required
      />
      <R.Fieldset
        legend="Expiry date"
        hint="For example, 12 2030"
      >
        <div className="sdr-date">
          <R.TextInput
            label="Month"
            name="card-exp-month"
            inputMode="numeric"
            autoComplete="cc-exp-month"
            defaultValue="12"
            pattern="0[1-9]|1[0-2]"
            maxLength={2}
            required
          />
          <R.TextInput
            label="Year"
            name="card-exp-year"
            inputMode="numeric"
            autoComplete="cc-exp-year"
            defaultValue="2030"
            pattern="20[0-9]{2}"
            maxLength={4}
            required
          />
        </div>
      </R.Fieldset>
      <R.TextInput
        label="Name on card"
        name="card-name"
        autoComplete="cc-name"
        defaultValue="ANA ȘERBAN"
        required
      />
      <R.TextInput
        label="Card security code"
        name="card-security"
        inputMode="numeric"
        autoComplete="cc-csc"
        defaultValue="123"
        pattern="[0-9]{3,4}"
        maxLength={4}
        hint="The 3 digits on the back, or 4 digits on the front, depending on your card."
        required
      />
      <div>
        <R.Button type="submit">Pay 25.00 RON</R.Button>
      </div>
      <p
        role="status"
        data-sdr-pattern-result
        tabIndex={-1}
        hidden={!done}
      >
        Your payment has been confirmed.
      </p>
    </form>
  );
}
~~~

## Service integration

Examples use library components. Server validation, messaging, payment processing and persistence are implemented by the application. Demonstration interactions stay local and do not persist after reload. JavaScript setup is documented once in [Get started](/en/getting-started/). Accessibility details are in [the accessibility guide](/en/accessibility/).
