~~~tsx
import { useState } from 'react';
import * as R from '@sdr/react';

const locale = 'en';

export function EligibilityExample() {
  const [answer, setAnswer] = useState('');
  return (
    <form
      data-sdr-pattern-demo="eligibility"
      className="sdr-stack"
      onSubmit={(event) => {
        event.preventDefault();
        const form = event.currentTarget;
        setAnswer(String(new FormData(form).get('resident')));
        requestAnimationFrame(() =>
          form
            .querySelector<HTMLElement>(
              '[data-sdr-pattern-outcome]:not([hidden])',
            )!
            .focus(),
        );
      }}
    >
      <R.InsetText>
        Illustrative conditions, not rules for a real service.
      </R.InsetText>
      <R.Fieldset legend="Do you live in the area where you are applying?">
        <R.Radio
          name="resident"
          value="yes"
          label="Yes"
          required
        />
        <R.Radio
          name="resident"
          value="no"
          label="No"
          required
        />
      </R.Fieldset>
      <div>
        <R.Button type="submit">Check</R.Button>
      </div>
      <div
        data-sdr-pattern-outcome="yes"
        tabIndex={-1}
        hidden={answer !== 'yes'}
      >
        <R.Alert
          tone="success"
          title="You can continue in this example"
        >
          <R.Link href={`/${locale}/patterns/start/`}>
            View required documents and steps
          </R.Link>
        </R.Alert>
      </div>
      <div
        data-sdr-pattern-outcome="no"
        tabIndex={-1}
        hidden={answer !== 'no'}
      >
        <R.Alert
          tone="warning"
          title="There is another option"
        >
          <R.Stack gap={3}>
            <p>You can get help finding the right institution.</p>
            <R.Link href={`/${locale}/patterns/contact/`}>
              View contact options
            </R.Link>
          </R.Stack>
        </R.Alert>
      </div>
    </form>
  );
}
~~~

## Service integration

Examples use library components. Server validation, messaging, payment processing and persistence are implemented by the application. Demonstration interactions stay local and do not persist after reload. JavaScript setup is documented once in [Get started](/en/getting-started/). Accessibility details are in [the accessibility guide](/en/accessibility/).
