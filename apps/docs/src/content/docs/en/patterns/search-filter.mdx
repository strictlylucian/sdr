## Example

Persist queries and filters in shareable URLs, announce result counts, and offer clear filters when nothing matches. Sorting must retain the chosen filters.

~~~tsx
import { useState } from 'react';
import * as R from '@sdr/react';

const locale = 'en';

export function SearchFilterExample() {
  const [filter, setFilter] = useState({ query: '', category: '' });
  const items = [
    { id: 'residence', label: 'Proof of residence', category: 'documents' },
    { id: 'copy', label: 'Document copy', category: 'documents' },
    { id: 'tax', label: 'Pay a fee', category: 'payments' },
  ];
  const visible = items.filter(
    (item) =>
      (!filter.category || item.category === filter.category) &&
      item.label
        .normalize('NFD')
        .replace(/\p{M}/gu, '')
        .toLowerCase()
        .includes(
          filter.query.normalize('NFD').replace(/\p{M}/gu, '').toLowerCase(),
        ),
  );
  return (
    <form
      data-sdr-pattern-demo="search"
      className="sdr-stack"
      onSubmit={(event) => {
        event.preventDefault();
        const data = new FormData(event.currentTarget);
        setFilter({
          query: String(data.get('query')),
          category: String(data.get('category')),
        });
      }}
      onReset={() => setFilter({ query: '', category: '' })}
    >
      <R.TextInput
        label="Find a service"
        type="search"
        name="query"
        autoComplete="off"
      />
      <R.Select
        label="Category"
        name="category"
      >
        <option value="">All categories</option>
        <option value="documents">Documents</option>
        <option value="payments">Payments</option>
      </R.Select>
      <R.Cluster>
        <R.Button type="submit">Search</R.Button>
        <R.Button
          variant="secondary"
          type="reset"
        >
          Clear filters
        </R.Button>
      </R.Cluster>
      <p
        role="status"
        data-sdr-search-count
      >{`Results: ${visible.length}`}</p>
      <p
        data-sdr-search-empty
        hidden={visible.length !== 0}
      >
        No results found. Try another search or clear the filters.
      </p>
      <R.Grid
        columns={2}
        data-sdr-search-results
        hidden={visible.length === 0}
      >
        {items.map((item) => (
          <div
            key={item.id}
            data-sdr-search-item
            data-category={item.category}
            data-search-label={item.label}
            hidden={!visible.includes(item)}
          >
            <R.Card
              title={item.label}
              footer={
                <R.Link href={`/${locale}/patterns/start/`}>View steps</R.Link>
              }
            >
              Example service.
            </R.Card>
          </div>
        ))}
      </R.Grid>
    </form>
  );
}
~~~

## Service integration

Examples use library components. Server validation, messaging, payment processing and persistence are implemented by the application. Demonstration interactions stay local and do not persist after reload. JavaScript setup is documented once in [Get started](/en/getting-started/). Accessibility details are in [the accessibility guide](/en/accessibility/).
