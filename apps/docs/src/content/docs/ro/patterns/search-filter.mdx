## Exemplu

URL-ul păstrează termenii de căutare și filtrele, astfel încât rezultatele să poată fi redeschise sau trimise altcuiva. Numărul de rezultate este anunțat și cititoarelor de ecran. Când nu există rezultate, filtrele pot fi șterse ușor. Sortarea păstrează filtrele selectate.

~~~tsx
import { useState } from 'react';
import * as R from '@strictlylucian/sdr-react';

const locale = 'ro';

export function SearchFilterExample() {
  const [filter, setFilter] = useState({ query: '', category: '' });
  const items = [
    {
      id: 'residence',
      label: 'Adeverință de domiciliu',
      category: 'documents',
    },
    { id: 'copy', label: 'Copie de document', category: 'documents' },
    { id: 'tax', label: 'Plata unei taxe', category: 'payments' },
  ];
  const visible = items.filter(
    (item) =>
      (!filter.category || item.category === filter.category) &&
      item.label
        .normalize('NFD')
        .replace(/\p{M}/gu, '')
        .toLowerCase()
        .includes(
          filter.query.normalize('NFD').replace(/\p{M}/gu, '').toLowerCase(),
        ),
  );
  return (
    <form
      data-sdr-pattern-demo="search"
      className="sdr-stack"
      onSubmit={(event) => {
        event.preventDefault();
        const data = new FormData(event.currentTarget);
        setFilter({
          query: String(data.get('query')),
          category: String(data.get('category')),
        });
      }}
      onReset={() => setFilter({ query: '', category: '' })}
    >
      <R.TextInput
        label="Caută un serviciu"
        type="search"
        name="query"
        autoComplete="off"
      />
      <R.Select
        label="Categorie"
        name="category"
      >
        <option value="">Toate categoriile</option>
        <option value="documents">Documente</option>
        <option value="payments">Plăți</option>
      </R.Select>
      <R.Cluster>
        <R.Button type="submit">Caută</R.Button>
        <R.Button
          variant="secondary"
          type="reset"
        >
          Șterge filtrele
        </R.Button>
      </R.Cluster>
      <p
        role="status"
        data-sdr-search-count
      >{`Rezultate: ${visible.length}`}</p>
      <p
        data-sdr-search-empty
        hidden={visible.length !== 0}
      >
        Nu am găsit rezultate. Încearcă alți termeni sau șterge filtrele.
      </p>
      <R.Grid
        columns={2}
        data-sdr-search-results
        hidden={visible.length === 0}
      >
        {items.map((item) => (
          <div
            key={item.id}
            data-sdr-search-item
            data-category={item.category}
            data-search-label={item.label}
            hidden={!visible.includes(item)}
          >
            <R.Card
              title={item.label}
              footer={
                <R.Link href={`/${locale}/patterns/start/`}>Vezi pașii</R.Link>
              }
            >
              Serviciu demonstrativ.
            </R.Card>
          </div>
        ))}
      </R.Grid>
    </form>
  );
}
~~~
