~~~tsx
import { useState } from 'react';
import * as R from '@strictlylucian/sdr-react';

const locale = 'en';

export function ValidationRecoveryExample() {
  const [done, setDone] = useState(false);
  return (
    <form
      data-sdr-pattern-demo="validation"
      className="sdr-stack"
      onSubmit={(event) => {
        event.preventDefault();
        const form = event.currentTarget;
        setDone(true);
        requestAnimationFrame(() =>
          form
            .closest('[data-sdr-pattern-demo]')!
            .querySelector<HTMLElement>('[data-sdr-pattern-result]')!
            .focus(),
        );
      }}
    >
      <div
        data-sdr-pattern-summary
        hidden={done}
      >
        <R.ErrorSummary
          errors={[
            { href: '#pattern-full-name', message: 'Enter your full name.' },
          ]}
          title="There is a problem"
        />
      </div>
      <R.TextInput
        id="pattern-full-name"
        name="full-name"
        label="Full name"
        autoComplete="name"
        required
        error={done ? undefined : 'Enter your full name.'}
        hint="Enter your full name."
      />
      <div>
        <R.Button type="submit">Continue</R.Button>
      </div>
      <p
        role="status"
        data-sdr-pattern-result
        tabIndex={-1}
        hidden={!done}
      >
        Your details have been checked. You can continue your application.
      </p>
    </form>
  );
}
~~~

## Service integration

Examples use library components. Server validation, messaging, payment processing and persistence are implemented by the application. Demonstration interactions stay local and do not persist after reload. JavaScript setup is documented once in [Get started](/en/getting-started/). Accessibility details are in [the accessibility guide](/en/accessibility/).
