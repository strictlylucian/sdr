~~~tsx
import { useState } from 'react';
import * as R from '@strictlylucian/sdr-react';

const locale = 'en';

export function EqualityInformationExample() {
  const [done, setDone] = useState(false);
  return (
    <form
      data-sdr-pattern-demo="equality"
      className="sdr-stack"
      onSubmit={(event) => {
        event.preventDefault();
        const form = event.currentTarget;
        setDone(true);
        requestAnimationFrame(() =>
          form
            .closest('[data-sdr-pattern-demo]')!
            .querySelector<HTMLElement>('[data-sdr-pattern-result]')!
            .focus(),
        );
      }}
    >
      <R.Fieldset
        legend="Age group (optional)"
        hint="An example monitoring question. Your answer does not affect eligibility."
      >
        <R.Radio
          name="age-group"
          value="under18"
          label="Under 18"
        />
        <R.Radio
          name="age-group"
          value="adult"
          label="18–64"
        />
        <R.Radio
          name="age-group"
          value="older"
          label="65 or older"
        />
        <R.Radio
          name="age-group"
          value="private"
          label="Prefer not to say"
        />
      </R.Fieldset>
      <div>
        <R.Button type="submit">Continue</R.Button>
      </div>
      <p
        role="status"
        data-sdr-pattern-result
        tabIndex={-1}
        hidden={!done}
      >
        Example completed. No details were stored.
      </p>
    </form>
  );
}
~~~

## Service integration

Examples use library components. Server validation, messaging, payment processing and persistence are implemented by the application. Demonstration interactions stay local and do not persist after reload. JavaScript setup is documented once in [Get started](/en/getting-started/). Accessibility details are in [the accessibility guide](/en/accessibility/).
