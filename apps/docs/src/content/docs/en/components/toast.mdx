~~~tsx
import { useState } from 'react';
import * as R from '@sdr/react';

const locale = 'en';

export function ToastExample() {
  const [show, setShow] = useState(false);
  const [generation, setGeneration] = useState(0);
  return (
    <R.Stack>
      <R.Button
        id="toast-trigger"
        data-sdr-toast-show
        onClick={() => {
          setGeneration((value) => value + 1);
          setShow(true);
        }}
      >
        Show confirmation
      </R.Button>
      <div
        hidden={!show}
        style={{ display: show ? 'contents' : undefined }}
        data-sdr-toast-example
      >
        <R.Toast
          key={generation}
          returnFocusId="toast-trigger"
          closeLabel="Close"
        >
          Changes saved.
        </R.Toast>
      </div>
    </R.Stack>
  );
}
~~~

## Usage

No auto-dismiss timer. Provide a real returnFocusId. Persistent errors belong in the page.



## API

| Property | Type | Required |
|---|---|---|
| `children` | `ReactNode` | Yes |
| `returnFocusId` | `string` | Yes |
| `closeLabel` | `string` | Yes |

Native HTML, aria-* and data-* attributes are available where allowed by the component types. Component-specific properties are extracted from TypeScript when generating documentation.
