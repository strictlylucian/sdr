~~~tsx
import { useState } from 'react';
import * as R from '@strictlylucian/sdr-react';

const locale = 'en';

export function FeedbackExample() {
  const [answered, setAnswered] = useState(false);
  return (
    <R.Stack>
      <R.Feedback
        question="Was this page helpful?"
        yesLabel="Yes"
        noLabel="No"
        onAnswer={() => setAnswered(true)}
      />
      <p role="status">{answered && 'Thank you for your feedback.'}</p>
    </R.Stack>
  );
}
~~~

## Usage

The host implements collection and privacy controls. Do not ask for personal data here.



## API

| Property | Type | Required |
|---|---|---|
| `question` | `string` | Yes |
| `yesLabel` | `string` | Yes |
| `noLabel` | `string` | Yes |
| `onAnswer` | `(helpful: boolean) => void` | Yes |

Native HTML, aria-* and data-* attributes are available where allowed by the component types. Component-specific properties are extracted from TypeScript when generating documentation.



## Accessibility
