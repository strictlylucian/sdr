                  lang: 'en',
                  label: 'English',
                  href: '/examples/en/page.html',
                  current: locale === 'en',

              ]}
            />

        </R.SdrProvider>

      navigation={
        <R.ServiceNavigation
          label="Main navigation"
          items={[
            { label: 'Request a document', href: '#page-main', current: true },

            { label: 'Help', href: '#page-help' },
          ]}
          menuLabel="Menu"
        />
      }
      beforeContent={
        <R.Breadcrumbs
          label="Breadcrumb"
          items={[
            { label: 'Services', href: '#page-main' },
            {
              label: 'Request a document',
              href: '#page-request',
              current: true,
            },
          ]}
        />
      }
      footer={
        <R.Stack gap={3}>
          <h2
            id="page-help"
            className="sdr-heading-sm"
          >
            Need help?
          </h2>
          <p>
            A real service provides contact details, opening hours and assisted
            support options here.
          </p>
          <R.Cluster>
            <R.Link href={`/${locale}/accessibility/`}>Accessibility</R.Link>
            <R.Link href={`/${locale}/patterns/cookies/`}>Cookies</R.Link>
          </R.Cluster>
        </R.Stack>
      }
    >
      <R.PageHeader
        className="sdr-u-padding-block-0"
        title="Request a document"
        description="Complete the details below and check your information before submitting."
      />
      <div className="sdr-grid-row">
        <section
          className="sdr-grid-column-two-thirds-from-desktop"
          id="page-request"
          aria-labelledby="page-request-title"
        >
          <R.Stack>
            <h2 id="page-request-title">Application details</h2>
            <form
              className="sdr-stack"
              data-sdr-page-form
              onSubmit={(event) => {
                event.preventDefault();
                setSubmitted(true);
                requestAnimationFrame(() =>
                  document.getElementById('page-confirmation')?.focus(),
                );
              }}
            >
              <R.TextInput
                id="full-name"
                name="full-name"
                label="Full name"
                autoComplete="name"
                required
              />
              <R.TextInput
                id="email"
                name="email"
                label="Email"
                type="email"
                autoComplete="email"
                hint="We will use this address to contact you about the application."
                required
              />
              <R.Select
                id="document-type"
                name="document-type"
                label="Document type"
                required
                defaultValue=""
              >
                <option value="">Choose a document</option>
                <option value="certificate">Certificate</option>
                <option value="copy">Document copy</option>
              </R.Select>
              <R.Checkbox
                id="confirm-details"
                name="confirm-details"
                label="I confirm that the details are correct."
                required
              />
              <div>
                <R.Button type="submit">Submit application</R.Button>
              </div>
            </form>
            <section
              id="page-confirmation"
              data-sdr-page-result
              hidden={!submitted}
              tabIndex={-1}
              aria-label="Application confirmation"
            >
              <R.Panel title="Application submitted">
                <p>
                  We have received your application. You will receive a response
                  by email.
                </p>
              </R.Panel>
            </section>
          </R.Stack>
        </section>
        <aside
          className="sdr-grid-column-one-third-from-desktop sdr-stack"
          aria-label="Service information"
        >
          <R.Card title="Before you start">
            <R.Stack gap={3}>
              <p>
                Have your identity document and supporting documents ready
                before you start.
              </p>
            </R.Stack>
          </R.Card>
          <section
            id="page-documents"
            aria-labelledby="page-documents-title"
          >
            <R.Stack gap={3}>
              <h2
                id="page-documents-title"
                className="sdr-heading-md"
              >
                Required documents
              </h2>
              <ul>
                <li>The applicant’s identity document.</li>
                <li>The original document, if you are requesting a copy.</li>
                <li>
                  Authorisation, if you are applying on someone else’s behalf.
                </li>
              </ul>
            </R.Stack>
          </section>
        </aside>
      </div>
    </R.Page>
  );
}

## Usage

Page provides full-width service chrome and a shared responsive content container. Header, navigation, back links or breadcrumbs, and footer content are supplied through slots. mainId identifies the skip-link target; skipLabel is supplied by the application. embedded uses a section for examples inside another page, with mainLabelledBy pointing to its heading. The host page body uses sdr-u-margin-0 to remove the browser’s default outer margin.



## API

| Property | Type | Required |
|---|---|---|
| `mainId` | `string` | Yes |
| `skipLabel` | `string` | Yes |
| `header` | `ReactNode` | Yes |
| `navigation` | `ReactNode` | No |
| `beforeContent` | `ReactNode` | No |
| `footer` | `ReactNode` | No |
| `embedded` | `boolean \| undefined` | No |
| `mainLabelledBy` | `string \| undefined` | No |

Native HTML, aria-* and data-* attributes are available where allowed by the component types. Component-specific properties are extracted from TypeScript when generating documentation.



## Accessibility

Composition accessibility depends on labels, element semantics, focus order and application context. Library themes and behaviours are checked against the documented examples; those results do not assess the consuming service. The matrix and limitations are described in [the accessibility guide](/en/accessibility/).
